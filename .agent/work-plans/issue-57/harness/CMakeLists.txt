cmake_minimum_required(VERSION 3.14)
project(s57_57_harness)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(geodesy REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(grid_map_ros REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_ros_common QUIET)
find_package(nav2_util QUIET)
find_package(rclcpp REQUIRED)
find_package(s57_msgs REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenMP)
set(HARNESS_SOURCES harness.cpp src/s57_layer.cpp today/s57_layer.cpp)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/jazzy/inflation_layer.cpp)
  list(APPEND HARNESS_SOURCES jazzy/inflation_layer.cpp)
endif()
add_executable(harness ${HARNESS_SOURCES})
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/jazzy/inflation_layer.cpp)
  target_compile_definitions(harness PRIVATE WITH_JAZZY_COPY)
endif()
target_include_directories(harness PRIVATE src ${CMAKE_CURRENT_SOURCE_DIR} ${grid_map_ros_INCLUDE_DIRS} ${geodesy_INCLUDE_DIRS})
target_link_libraries(harness
  ${geometry_msgs_TARGETS} ${grid_map_ros_LIBRARIES}
  nav2_costmap_2d::layers nav2_costmap_2d::nav2_costmap_2d_core
  rclcpp::rclcpp ${s57_msgs_TARGETS} tf2_geometry_msgs::tf2_geometry_msgs tf2_ros::tf2_ros
  ${geodesy_LIBRARIES})
if(nav2_ros_common_FOUND)
  target_link_libraries(harness nav2_ros_common::nav2_ros_common)
else()
  target_link_libraries(harness nav2_util::nav2_util_core)
endif()
